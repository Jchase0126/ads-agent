"""离线动态探针：真的执行 panel 的 _sync_composer_status，看状态栏文案。"""
import inspect, sys, types
sys.path.insert(0, r"D:/Antenna/ADS Agent/addon/ads_agent")
import panel

class _W:
    def __init__(self, text=""):
        self._t = text; self.tooltip = None; self.acc = None
        self.visible = None; self.enabled = None
    def text(self): return self._t
    def setText(self, v): self._t = v
    def setToolTip(self, v): self.tooltip = v
    def setAccessibleDescription(self, v): self.acc = v
    def setVisible(self, v): self.visible = v
    def setEnabled(self, v): self.enabled = v

class _Send(_W):
    busy = False
    def property(self, name): return self.busy

def _owner():
    for name, obj in vars(panel).items():
        if inspect.isclass(obj) and "_sync_composer_status" in vars(obj):
            return name, obj
    raise SystemExit("找不到 _sync_composer_status 所在类")

def render(tool_name, raw):
    cls_name, cls = _owner()
    self_ = types.SimpleNamespace(
        status=_W(raw), send=_Send(), projects_data={"active": "p1"},
        _turn_project="p1", _tool_name=tool_name, _stopping=False,
        run_status_row=_W(), run_status_text=_W(),
        run_status_dot=_W(), run_stop_btn=_W(),
    )
    cls._sync_composer_status(self_)
    return cls_name, self_.run_status_text.text()

for tool, raw in [("create_workspace", "执行 create_workspace 已用 3s"),
                  ("list_workspaces", "执行 list_workspaces"),
                  ("some_unknown_tool", "执行 some_unknown_tool")]:
    print(render(tool, raw))
